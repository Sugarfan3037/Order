public class Order {
	private String name;
	private int total = 0;
    private int redQty=0;
    private int greenQty=0;

    public Order(String name,int total,int redQty,int greenQty,int milkteaQty,int hamQty,int friQty,int chiQty,int sandQty)
    {
    super();
    this.total=total;
    this.redQty=redQty;
    this.greenQty=greenQty;
    this.milkteaQty=milkteaQty;
    this.hamQty=hamQty;
    this.friQty=friQty;
    this.chiQty=chiQty;
    this.sandQty=sandQty;
    total=redQty*20+greenQty*20+milkteaQty*30+hamQty*60+friQty*50+chiQty*50+sandQty*80;
    }
    
    
	public int getTotal() {
		return total;
	}
	public void setTotal(int total) {
		this.total = total;
	}
	public int getRedQty() {
		return redQty;
	}
	public void setRedQty(int redQty) {
		this.redQty = redQty;
	}
	public int getGreenQty() {
		return greenQty;
	}
	public void setGreenQty(int greenQty) {
		this.greenQty = greenQty;
	}
	public int getMilkteaQty() {
		return milkteaQty;
	}
	public void setMilkteaQty(int milkteaQty) {
		this.milkteaQty = milkteaQty;
	}
	public int getHamQty() {
		return hamQty;
	}
	public void setHamQty(int hamQty) {
		this.hamQty = hamQty;
	}
	public int getFriQty() {
		return friQty;
	}
	public void setFriQty(int friQty) {
		this.friQty = friQty;
	}
	public int getChiQty() {
		return chiQty;
	}
	public void setChiQty(int chiQty) {
		this.chiQty = chiQty;
	}
	public int getSandQty() {
		return sandQty;
	}
	public void setSandQty(int sandQty) {
		this.sandQty = sandQty;
	}
	public void show()
    {
    	System.out.println("總價:"+total);
    	if(member)
    	{
    		total=(int) (total*0.9);
    	}
    }
    
}

	private void updateDetail(JTextArea textArea_1)
	{
		String detail="";
		if(redQty>0)
		{
			detail+="紅茶x"+redQty+"="+redQty*20+"元"+"\n";
		}
		if(greenQty>0)
		{
			detail+="綠茶x"+greenQty+"="+greenQty*20+"元"+"\n";
		}
		if(milkteaQty>0)
		{
			detail+="奶茶x"+milkteaQty+"="+milkteaQty*30+"元"+"\n";
		}
		if(hamQty>0)
		{
			detail+="漢堡x"+hamQty+"="+hamQty*60+"元"+"\n";
		}
		if(friQty>0)
		{
			detail+="薯條x"+friQty+"="+friQty*50+"元"+"\n";
		}
		if(chiQty>0)
		{
			detail+="雞塊x"+chiQty+"="+chiQty*50+"元"+"\n";
		}
		if(sandQty>0)
		{
			detail+="總匯x"+sandQty+"="+sandQty*80+"元"+"\n";
		}
		textArea_1.setText(detail);
		
	}
